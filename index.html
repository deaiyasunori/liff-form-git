const CHANNEL_ACCESS_TOKEN = "ここに取得したチャネルアクセストークンを貼り付け"; // LINEからの通知（Webhook）を受け取る関数 function doPost(e) { const json = JSON.parse(e.postData.contents); const event = json.events[0]; if (!event) return; // 1. 通知を送ってきたユーザーの LINE ユーザーID を取得 const userId = event.source.userId; const replyToken = event.replyToken; // 2. LINE APIを叩いて、ユーザーの「表示名（お名前）」を取得する const userName = getUserName(userId); // 3. メッセージの内容に応じた処理（例: 「申込み」と送信された場合） if (event.type === 'message' && event.message.text === '申込み') { // スプレッドシートに記録（例: 日時, ユーザーID, お名前） const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName('シート1') || ss.getSheets()[0]; sheet.appendRow([new Date(), userId, userName, "申込み完了"]); // ユーザーに確認メッセージを返信 replyMessage(replyToken, `${userName} 様\nお申込みを受け付けました！`); } return ContentService.createTextOutput(JSON.stringify({ content: 'ok' })).setMimeType(ContentService.MimeType.JSON); } // ユーザーIDからLINE表示名を取得する関数 function getUserName(userId) { const url = `https://api.line.me/v2/bot/profile/${userId}`; const response = UrlFetchApp.fetch(url, { headers: { 'Authorization': `Bearer ${CHANNEL_ACCESS_TOKEN}` }, muteHttpExceptions: true }); if (response.getResponseCode() === 200) { const data = JSON.parse(response.getContentText()); return data.displayName; // LINEのお名前 } return "ゲスト"; } // 応答メッセージを送る関数 function replyMessage(replyToken, text) { const url = 'https://api.line.me/v2/bot/message/reply'; UrlFetchApp.fetch(url, { headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${CHANNEL_ACCESS_TOKEN}` }, method: 'post', payload: JSON.stringify({ replyToken: replyToken, messages: [{ type: 'text', text: text }] }) }); }
